## probe file (only on the probe branch)
package platform

// R71 Gate 3 fault-injection probes. This file exists only on the
// probe/r71-g3-fault-injection branch and is never merged. Every probe is
// inert unless its own environment variable is set.

import (
	"os"
	"strconv"
	"strings"
	"time"
)

// probeCloseDelay simulates worker-side close_session latency. It is read on
// every worker close call so the A and B arms share the same injected delay.
func probeCloseDelay() time.Duration {
	setting := strings.TrimSpace(os.Getenv("PERI_R71_G3_CLOSE_DELAY_MS"))
	if setting == "" {
		return 0
	}
	milliseconds, err := strconv.Atoi(setting)
	if err != nil || milliseconds <= 0 {
		return 0
	}
	return time.Duration(milliseconds) * time.Millisecond
}

// probeDropCloseRecord drops the durable close record (operation and audit)
// while the client still receives the success reply. It deterministically
// reproduces a power loss that discards the tail record.
func probeDropCloseRecord() bool {
	return strings.TrimSpace(os.Getenv("PERI_R71_G3_DROP_CLOSE_RECORD")) == "1"
}

// probeOperationQueueMaximum lowers the operation journal admission bound so a
// settlement can meet a full queue on purpose.
func probeOperationQueueMaximum() int {
	setting := strings.TrimSpace(os.Getenv("PERI_R71_G3_OPERATION_QUEUE_MAX"))
	if setting == "" {
		return 0
	}
	limit, err := strconv.Atoi(setting)
	if err != nil || limit <= 0 {
		return 0
	}
	return limit
}
